set(BUILD_SHARED_LIBS ON)
set(ENABLE_TEST OFF)
set(ENABLE_WERROR OFF)
set(ENABLE_INSTALL ON)

set(ENABLE_TAR_SHARED ON)
set(ENABLE_CPIO_SHARED ON)
set(ENABLE_CAT_SHARED ON)
set(ENABLE_UNZIP_SHARED ON)

set(ENABLE_LZ4 OFF)
set(ENABLE_LZO OFF)
set(ENABLE_LIBB2 OFF)
set(ENABLE_LIBXML2 OFF)
set(ENABLE_PCREPOSIX OFF)
set(ENABLE_PCRE2POSIX OFF)
set(ENABLE_LIBGCC OFF)
set(ENABLE_CNG OFF)
set(ENABLE_MBEDTLS OFF)
set(ENABLE_NETTLE OFF)
set(ENABLE_WIN32_XMLLITE OFF)

set(ENABLE_ACL OFF)
set(ENABLE_XATTR OFF)

set(ENABLE_ZLIB ON)
set(CMAKE_DISABLE_FIND_PACKAGE_ZLIB TRUE)
set(ZLIB_FOUND TRUE)
set(ZLIB_INCLUDE_DIR ${CMAKE_SOURCE_DIR}/ports/zlib/repo)
set(ZLIB_LIBRARIES zlib)

set(ENABLE_BZip2 ON)
set(CMAKE_DISABLE_FIND_PACKAGE_BZip2 TRUE)
set(BZIP2_FOUND TRUE)
set(BZIP2_INCLUDE_DIR ${CMAKE_SOURCE_DIR}/ports/bzip2/repo)
set(BZIP2_LIBRARIES bzip2)

set(ENABLE_LZMA ON)
set(CMAKE_DISABLE_FIND_PACKAGE_LibLZMA TRUE)
set(LIBLZMA_FOUND TRUE)
set(LIBLZMA_INCLUDE_DIRS ${CMAKE_SOURCE_DIR}/ports/xz/repo/src/liblzma/api)
set(LIBLZMA_LIBRARIES liblzma)
set(HAVE_LZMA_STREAM_ENCODER_MT 1 CACHE INTERNAL "")

set(ENABLE_ZSTD ON)
set(ZSTD_INCLUDE_DIR ${CMAKE_SOURCE_DIR}/ports/zstd/repo/lib CACHE PATH "" FORCE)
set(ZSTD_LIBRARY zstd CACHE FILEPATH "" FORCE)
set(HAVE_LIBZSTD 1 CACHE INTERNAL "")
set(HAVE_ZSTD_compressStream 1 CACHE INTERNAL "")
set(HAVE_ZSTD_minCLevel 1 CACHE INTERNAL "")

set(ENABLE_OPENSSL ON)
set(CMAKE_DISABLE_FIND_PACKAGE_OpenSSL TRUE)
set(OPENSSL_FOUND TRUE)
set(OPENSSL_INCLUDE_DIR ${CMAKE_BINARY_DIR}/ports/openssl/gen/include)
set(OPENSSL_CRYPTO_LIBRARY "")
set(OPENSSL_LIBRARIES "")
set(HAVE_OPENSSL_EVP_H 1 CACHE INTERNAL "")
set(HAVE_OPENSSL_OPENSSLV_H 1 CACHE INTERNAL "")
set(HAVE_PKCS5_PBKDF2_HMAC_SHA1 1 CACHE INTERNAL "")
foreach(_alg MD5 RMD160 SHA1 SHA256 SHA384 SHA512)
    set(ARCHIVE_CRYPTO_${_alg}_OPENSSL TRUE CACHE INTERNAL "")
endforeach()

set(ENABLE_ICONV ON)
set(ICONV_INCLUDE_DIR ${CMAKE_BINARY_DIR}/ports/libiconv/gen/include CACHE PATH "" FORCE)
set(LIBICONV_PATH libiconv CACHE FILEPATH "" FORCE)
set(LIBCHARSET_PATH libiconv CACHE FILEPATH "" FORCE)
set(HAVE_ICONV_libc_const "" CACHE INTERNAL "")
set(HAVE_ICONV_libc_ "" CACHE INTERNAL "")
set(HAVE_ICONV_libiconv_const "" CACHE INTERNAL "")
set(HAVE_ICONV_libiconv_ 1 CACHE INTERNAL "")
set(HAVE_LOCALCHARSET_H 1 CACHE INTERNAL "")
set(HAVE_LOCALE_CHARSET 1 CACHE INTERNAL "")

set(ENABLE_EXPAT ON)
set(CMAKE_DISABLE_FIND_PACKAGE_EXPAT TRUE)
set(EXPAT_FOUND TRUE)
set(EXPAT_INCLUDE_DIR ${CMAKE_SOURCE_DIR}/ports/expat/repo/expat/lib)
set(EXPAT_LIBRARIES expat)
set(HAVE_EXPAT_H 1 CACHE INTERNAL "")

set(CMAKE_TRY_COMPILE_TARGET_TYPE EXECUTABLE)

set(HAVE__FSEEKI64 0 CACHE INTERNAL "")

set(HAVE_LINUX_FIEMAP_H 0 CACHE INTERNAL "")
set(HAVE_LINUX_FS_H 0 CACHE INTERNAL "")
set(HAVE_LINUX_MAGIC_H 0 CACHE INTERNAL "")
set(HAVE_LINUX_TYPES_H 0 CACHE INTERNAL "")

set(CMAKE_MESSAGE_LOG_LEVEL NOTICE)
add_subdirectory(repo)

set(LIBARCHIVE_CFLAGS
    ${USERSPACE_CFLAGS}
    -Wno-sign-conversion
    -Wno-conversion
    -Wno-unused-function
    -Wno-unused-variable
    -Wno-unused-parameter
)

target_compile_options(archive PRIVATE ${LIBARCHIVE_CFLAGS} -fvisibility=default)

foreach(_target archive archive_static)
    target_include_directories(${_target} SYSTEM PRIVATE ${CMAKE_SOURCE_DIR}/ports/openssl/repo/include)
    target_link_libraries(${_target} libcrypto)
endforeach()

foreach(_prog bsdtar bsdcat bsdcpio bsdunzip)
    target_compile_options(${_prog} PRIVATE ${LIBARCHIVE_CFLAGS})
endforeach()

add_custom_target(libarchive)
add_dependencies(libarchive
    archive
    bsdcat
    bsdcpio
    bsdtar
    bsdunzip
)
